    46→  start(): void {
    47→    if (this.running) return
    48→    this.running = true
    49→    this.settings = loadSettings()
    50→    this.settingsReadAt = Date.now()
    51→
    52→    // ---- startup resume ----
    53→    try {
    54→      run("UPDATE chapters SET status = 'scraped' WHERE status = 'translating'")
    55→      const runningJobs = all<{ id: string }>("SELECT id FROM jobs WHERE status = 'running'")
    56→      for (const j of runningJobs) {
    57→        run('UPDATE jobs SET status = ?, updated_at = ? WHERE id = ?', 'queued', now(), j.id)
    58→      }
    59→      const queued = all<JobRowDb>("SELECT * FROM jobs WHERE status = 'queued' ORDER BY created_at ASC")
    60→      for (const job of queued) {
    61→        try {
    62→          this.reenqueueJob(job)
    63→        } catch (e) {
    64→          logError(`Không enqueue lại job ${job.id}: ${(e as Error).message}`)
    65→        }
    66→      }
    67→      logInfo(`Engine online — tiếp tục ${queued.length} job đang chờ`)
    68→    } catch (e) {
    69→      logError(`Lỗi resume khi khởi động: ${(e as Error).message}`)
    70→    }
    71→
    72→    // ---- worker pools ----
    73→    for (let i = 0; i < MAX_WORKERS; i++) void this.scrapeWorkerLoop(i)
    74→    for (let i = 0; i < MAX_WORKERS; i++) void this.translateWorkerLoop(i)
    75→
    76→    // ---- stats broadcast ----
    77→    this.statsTimer = setInterval(() => {
    78→      try {
    79→        emitStats(this.buildStats())
    80→      } catch {
    81→        /* ignore */
    82→      }
    83→    }, 3000)
    84→  }

    98→  /** POST /jobs/scrape — sync fetch meta + upsert novel/chapters, enqueue. */
    99→  async startScrapeJob(body: Record<string, unknown>): Promise<{ jobId: string; novelId: string; totalChapters: number; tasks: number }> {
   100→    const url = String(body?.url ?? '').trim()
   101→    if (!url) throw new Error('Thiếu tham số url')
   102→    const adapter = detectAdapter(url)
   103→    if (!adapter) {
   104→      let host = ''
   105→      try {
   106→        host = new URL(url).hostname
   107→      } catch {
   108→        /* ignore */
   109→      }
   110→      throw new Error(
   111→        `Không hỗ trợ site ${host || url} — hiện hỗ trợ: 69shuba.com, xbiquge.info, novel543.com, hjwzw.com, truyenfull.live`
   112→      )
   113→    }
   114→    const bookUrl = adapter.normalizeUrl(url)
   115→
   116→    logInfo(`Đang tải thông tin truyện từ ${adapter.siteName}: ${bookUrl}`)
   117→    const { meta, chapters } = await adapter.fetchBook(bookUrl)
   118→    if (!chapters.length) throw new Error('Không tìm thấy chương nào cho truyện này')
   119→    const ts = now()
   120→
   121→    // ---- upsert novel by source_url ----
   122→    let novel = get<NovelRow>('SELECT * FROM novels WHERE source_url = ?', bookUrl)
   123→    let addedChapters = 0
   124→    if (!novel) {
   125→      const novelId = uuid()
   126→      run(
   127→        `INSERT INTO novels (id, source_site, source_url, title, author, cover_url, description, lang, status, total_chapters, scraped_count, translated_count, created_at, updated_at)
   128→         VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?)`,
   129→        novelId,
   130→        adapter.id,
   131→        bookUrl,
   132→        meta.title,
   133→        meta.author,
   134→        meta.cover,
   135→        meta.description,
   136→        meta.lang,
   137→        meta.status,
   138→        chapters.length,
   139→        0,
   140→        0,
   141→        ts,
   142→        ts
   143→      )
   144→      const stmt = (globalThis as any).__engineDb!.prepare(
   145→        `INSERT INTO chapters (id, novel_id, idx, title, source_url, status) VALUES (?,?,?,?,?, 'pending')`
   146→      )
   147→      chapters.forEach((c, i) => stmt.run(uuid(), novelId, i + 1, c.title, c.url))
   148→      novel = getNovel(novelId)!
   149→      logSuccess(`Đã thêm truyện mới «${meta.title}» (${chapters.length} chương)`)
   150→    } else {
   151→      run(
   152→        `UPDATE novels SET title=?, author=?, cover_url=?, description=?, status=?, total_chapters=?, updated_at=? WHERE id=?`,
   153→        meta.title,
   154→        meta.author ?? novel.author,
   155→        meta.cover ?? novel.cover_url,
   156→        meta.description ?? novel.description,
   157→        meta.status ?? novel.status,
   158→        chapters.length,
   159→        ts,
   160→        novel.id
   161→      )
   162→      // append chapters not yet known (match by source_url)
   163→      const existing = all<{ idx: number; source_url: string }>(
   164→        'SELECT idx, source_url FROM chapters WHERE novel_id = ?',
   165→        novel.id
   166→      )
   167→      const knownUrls = new Set(existing.map((e) => e.source_url))
   168→      const takenIdx = new Set<number>(existing.map((e) => e.idx))
   169→      let appendCounter = Math.max(chapters.length, ...takenIdx, 0)
   170→      const stmt = (globalThis as any).__engineDb!.prepare(
   171→        `INSERT INTO chapters (id, novel_id, idx, title, source_url, status) VALUES (?,?,?,?,?, 'pending')`
   172→      )
   173→      for (let i = 0; i < chapters.length; i++) {
   174→        const c = chapters[i]
   175→        if (knownUrls.has(c.url)) continue
   176→        let idx = i + 1
   177→        if (takenIdx.has(idx)) idx = ++appendCounter
   178→        takenIdx.add(idx)
   179→        stmt.run(uuid(), novel.id, idx, c.title, c.url)
   180→        addedChapters++
   181→      }
   182→      novel = getNovel(novel.id)!
   183→      if (addedChapters > 0) logInfo(`Đã thêm ${addedChapters} chương mới cho «${meta.title}»`)
   184→    }
   185→
   186→    // ---- select tasks for this job ----
   187→    const from = body?.chapterFrom != null ? Math.max(1, parseInt(String(body.chapterFrom), 10) || 1) : null
   188→    const to = body?.chapterTo != null ? parseInt(String(body.chapterTo), 10) || null : null
   189→    let rows = all<ChapterRow>('SELECT * FROM chapters WHERE novel_id = ? ORDER BY idx ASC', novel.id)
   190→    if (from != null || to != null) {
   191→      rows = rows.filter((r) => (!from || r.idx >= from) && (!to || r.idx <= to))
   192→    }
   193→    const taskRows = rows.filter((r) => r.content === null || r.status === 'pending' || r.status === 'error')
   194→
   195→    const jobId = uuid()
   196→    const params = {
   197→      url: bookUrl,
   198→      autoTranslate: body?.autoTranslate === true,
   199→      chapterFrom: from,
   200→      chapterTo: to,
   201→    }
   202→    run(
   203→      `INSERT INTO jobs (id, type, status, novel_id, params, progress, total, created_at, updated_at)
   204→       VALUES (?,?, 'queued', ?, ?, 0, ?, ?, ?)`,
   205→      jobId,
   206→      'scrape',
   207→      novel.id,
   208→      JSON.stringify(params),
   209→      taskRows.length,
   210→      ts,
   211→      ts
   212→    )
   213→    logInfo(
   214→      `Đã tạo job cào «${novel.title}»: ${taskRows.length}/${rows.length} chương cần cào${addedChapters ? ` (trong đó ${addedChapters} chương mới)` : ''}`
   215→    )
   216→    emitJob(jobId)
   217→    emitNovel(novelRowToSummary(novel))
   218→    this.enqueueScrapeTasks(jobId, novel.id, taskRows)
   219→    if (taskRows.length === 0) this.maybeFinishJob(jobId)
   220→    return { jobId, novelId: novel.id, totalChapters: rows.length, tasks: taskRows.length }
   221→  }

   223→  /** POST /jobs/translate */
   224→  startTranslateJob(body: Record<string, unknown>): { jobId: string; novelId: string; note?: string } {
   225→    const novelId = String(body?.novelId ?? '').trim()
   226→    if (!novelId) throw new Error('Thiếu tham số novelId')
   227→    const novel = getNovel(novelId)
   228→    if (!novel) throw new Error('Không tìm thấy truyện')
   229→    const force = body?.force === true
   230→    const s = this.getSettings()
   231→    const ts = now()
   232→
   233→    const jobId = uuid()
   234→    const params = { novelId, force }
   235→
   236→    if (!hasActiveGeminiKeys()) {
   237→      run(
   238→        `INSERT INTO jobs (id, type, status, novel_id, params, progress, total, error, created_at, updated_at, finished_at)
   239→         VALUES (?,?, 'error', ?, ?, 0, 0, ?, ?, ?, ?)`,
   240→        jobId,
   241→        'translate',
   242→        novelId,
   243→        JSON.stringify(params),
   244→        'Chưa có Gemini API key hoạt động — thêm key trong Cài đặt',
   245→        ts,
   246→        ts,
   247→        ts
   248→      )
   249→      logWarn('Không thể tạo job dịch: chưa có Gemini API key hoạt động')
   250→      emitJob(jobId)
   251→      return { jobId, novelId, note: 'Chưa có Gemini API key hoạt động — thêm key trong Cài đặt' }
   252→    }
   253→
   254→    const srcLang = s.sourceLang === 'auto' ? novel.lang : s.sourceLang
   255→    if (srcLang === s.targetLang) {
   256→      run(
   257→        `INSERT INTO jobs (id, type, status, novel_id, params, progress, total, created_at, updated_at, finished_at)
   258→         VALUES (?,?, 'done', ?, ?, 0, 0, ?, ?, ?, ?)`,
   259→        jobId,
   260→        'translate',
   261→        novelId,
   262→        JSON.stringify(params),
   263→        ts,
   264→        ts,
   265→        ts
   266→      )
   267→      logInfo(`Bỏ qua dịch «${novel.title}»: ngôn ngữ nguồn (${srcLang}) đã là ngôn ngữ đích (${s.targetLang})`)
   268→      emitJob(jobId)
   269→      return { jobId, novelId, note: `Bỏ qua dịch: nguồn đã là ${srcLang}` }
   270→    }
   271→
   272→    const rows = all<ChapterRow>(
   273→      `SELECT * FROM chapters WHERE novel_id = ? AND content IS NOT NULL ${force ? '' : 'AND translated IS NULL'} ORDER BY idx ASC`,
   274→      novelId
   275→    )
   276→    run(
   277→      `INSERT INTO jobs (id, type, status, novel_id, params, progress, total, created_at, updated_at)
   278→       VALUES (?,?, 'queued', ?, ?, 0, ?, ?, ?)`,
   279→      jobId,
   280→      'translate',
   281→      novelId,
   282→      JSON.stringify(params),
   283→      rows.length,
   284→      ts,
   285→      ts
   286→    )
   287→    logInfo(`Đã tạo job dịch «${novel.title}»: ${rows.length} chương${force ? ' (dịch lại tất cả)' : ''}`)
   288→    emitJob(jobId)
   289→    this.enqueueTranslateTasks(jobId, novelId, rows)
   290→    if (rows.length === 0) this.maybeFinishJob(jobId)
   291→    return { jobId, novelId }
   292→  }

   328→  pause(jobId: string, reason?: string): void {
   329→    const job = getJob(jobId)
   330→    if (!job) throw new Error('Không tìm thấy job')
   331→    if (job.status !== 'running' && job.status !== 'queued') {
   332→      throw new Error(`Không thể tạm dừng job đang ${job.status}`)
   333→    }
   334→    run('UPDATE jobs SET status = ?, error = COALESCE(?, error), updated_at = ? WHERE id = ?', 'paused', reason ?? null, now(), jobId)
   335→    this.dropQueuedTasks(jobId)
   336→    this.consecutiveErrors.delete(jobId)
   337→    logInfo(`Đã tạm dừng job ${jobId.slice(0, 8)} (${job.type})${reason ? ` — ${reason}` : ''}`)
   338→    emitJob(jobId)
   339→  }
   340→
   341→  resume(jobId: string): void {
   342→    const job = getJob(jobId)
   343→    if (!job) throw new Error('Không tìm thấy job')
   344→    if (job.status !== 'paused') throw new Error(`Không thể tiếp tục job đang ${job.status}`)
   345→    run('UPDATE jobs SET status = ?, updated_at = ? WHERE id = ?', 'queued', now(), jobId)
   346→    this.consecutiveErrors.delete(jobId)
   347→    this.reenqueueJob(job)
   348→    logInfo(`Đã tiếp tục job ${jobId.slice(0, 8)} (${job.type})`)
   349→    emitJob(jobId)
   350→  }
   351→
   352→  cancel(jobId: string): void {
   353→    const job = getJob(jobId)
   354→    if (!job) throw new Error('Không tìm thấy job')
   355→    if (job.status === 'done' || job.status === 'cancelled') {
   356→      this.dropQueuedTasks(jobId)
   357→      return
   358→    }
   359→    run('UPDATE jobs SET status = ?, finished_at = ?, updated_at = ? WHERE id = ?', 'cancelled', now(), now(), jobId)
   360→    this.dropQueuedTasks(jobId)
   361→    this.consecutiveErrors.delete(jobId)
   362→    logInfo(`Đã huỷ job ${jobId.slice(0, 8)} (${job.type})`)
   363→    emitJob(jobId)
   364→  }

   366→  private dropQueuedTasks(jobId: string): void {
   367→    this.scrapeQueue = this.scrapeQueue.filter((t) => t.jobId !== jobId)
   368→    this.translateQueue = this.translateQueue.filter((t) => t.jobId !== jobId)
   369→  }

   371→  /** Recompute remaining tasks of a (paused/queued) job FROM the DB. */
   372→  private reenqueueJob(job: JobRowDb): void {
   373→    let params: Record<string, unknown> = {}
   374→    try {
   375→      params = JSON.parse(job.params || '{}')
   376→    } catch {
   377→      params = {}
   378→    }
   379→    const novelId = job.novel_id
   380→    if (!novelId) return
   381→
   382→    let taskRows: ChapterRow[] = []
   383→    if (job.type === 'translate') {
   384→      const force = params.force === true
   385→      taskRows = all<ChapterRow>(
   386→        `SELECT * FROM chapters WHERE novel_id = ? AND content IS NOT NULL ${force ? "AND status != 'translated'" : 'AND translated IS NULL'} ORDER BY idx ASC`,
   387→        novelId
   388→      )
   389→      this.enqueueTranslateTasks(job.id, novelId, taskRows)
   390→    } else {
   391→      const from = params.chapterFrom != null ? Number(params.chapterFrom) : null
   392→      const to = params.chapterTo != null ? Number(params.chapterTo) : null
   393→      let sql = "SELECT * FROM chapters WHERE novel_id = ? AND (content IS NULL OR status IN ('pending','error'))"
   394→      const args: unknown[] = [novelId]
   395→      if (from != null && to != null) {
   396→        sql += ' AND idx BETWEEN ? AND ?'
   397→        args.push(from, to)
   398→      } else if (from != null) {
   399→        sql += ' AND idx >= ?'
   400→        args.push(from)
   401→      } else if (to != null) {
   402→        sql += ' AND idx <= ?'
   403→        args.push(to)
   404→      }
   405→      sql += ' ORDER BY idx ASC'
   406→      taskRows = all<ChapterRow>(sql, ...args)
   407→      this.enqueueScrapeTasks(job.id, novelId, taskRows)
   408→    }
   409→    // keep totals consistent: progress + remaining
   410→    run('UPDATE jobs SET total = ?, updated_at = ? WHERE id = ?', job.progress + taskRows.length, now(), job.id)
   411→    if (taskRows.length === 0) this.maybeFinishJob(job.id)
   412→  }
   413→
   414→  private enqueueScrapeTasks(jobId: string, novelId: string, rows: ChapterRow[]): void {
   415→    for (const r of rows) {
   416→      this.scrapeQueue.push({ jobId, novelId, chapterId: r.id })
   417→    }
   418→  }
   419→
   420→  private enqueueTranslateTasks(jobId: string, novelId: string, rows: ChapterRow[]): void {
   421→    for (const r of rows) {
   422→      this.translateQueue.push({ jobId, novelId, chapterId: r.id })
   423→    }
   424→  }

   430→  private async scrapeWorkerLoop(_workerId: number): Promise<void> {
   431→    while (this.running) {
   432→      const s = this.getSettings()
   433→      const concurrency = Math.max(1, Math.min(s.scrapeConcurrency || 3, MAX_WORKERS))
   434→      if (this.scrapeActive >= concurrency) {
   435→        await sleep(250)
   436→        continue
   437→      }
   438→      const task = this.scrapeQueue.shift()
   439→      if (!task) {
   440→        await sleep(300)
   441→        continue
   442→      }
   443→      this.scrapeActive++
   444→      this.inFlight.set(task.jobId, (this.inFlight.get(task.jobId) ?? 0) + 1)
   445→      try {
   446→        await this.processScrapeTask(task)
   447→      } catch (e) {
   448→        logError(`Lỗi worker cào (job ${task.jobId.slice(0, 8)}): ${(e as Error).message}`)
   449→      } finally {
   450→        this.scrapeActive--
   451→        const v = (this.inFlight.get(task.jobId) ?? 1) - 1
   452→        if (v <= 0) this.inFlight.delete(task.jobId)
   453→        else this.inFlight.set(task.jobId, v)
   454→        try {
   455→          this.maybeFinishJob(task.jobId)
   456→        } catch {
   457→          /* ignore */
   458→        }
   459→      }
   460→    }
   461→  }
   462→
   463→  private async translateWorkerLoop(_workerId: number): Promise<void> {
   464→    while (this.running) {
   465→      const s = this.getSettings()
   466→      const concurrency = Math.max(1, Math.min(s.translateConcurrency || 2, MAX_WORKERS))
   467→      if (this.translateActive >= concurrency) {
   468→        await sleep(250)
   469→        continue
   470→      }
   471→      const task = this.translateQueue.shift()
   472→      if (!task) {
   473→        await sleep(300)
   474→        continue
   475→      }
   476→      this.translateActive++
   477→      this.inFlight.set(task.jobId, (this.inFlight.get(task.jobId) ?? 0) + 1)
   478→      try {
   479→        await this.processTranslateTask(task)
   480→      } catch (e) {
   481→        logError(`Lỗi worker dịch (job ${task.jobId.slice(0, 8)}): ${(e as Error).message}`)
   482→      } finally {
   483→        this.translateActive--
   484→        const v = (this.inFlight.get(task.jobId) ?? 1) - 1
   485→        if (v <= 0) this.inFlight.delete(task.jobId)
   486→        else this.inFlight.set(task.jobId, v)
   487→        try {
   488→          this.maybeFinishJob(task.jobId)
   489→        } catch {
   490→          /* ignore */
   491→        }
   492→      }
   493→    }
   494→  }

   496→  private ensureJobRunning(jobId: string): boolean {
   497→    const job = getJob(jobId)
   498→    if (!job) return false
   499→    if (job.status === 'paused' || job.status === 'cancelled' || job.status === 'done' || job.status === 'error') {
   500→      return false
   501→    }
   502→    if (job.status === 'queued') {
   503→      run('UPDATE jobs SET status = ?, updated_at = ? WHERE id = ?', 'running', now(), jobId)
   504→      emitJob(jobId)
   505→    }
   506→    return true
   507→  }
   508→
   509→  private async processScrapeTask(task: ScrapeTask): Promise<void> {
   510→    if (!this.ensureJobRunning(task.jobId)) return
   511→    const s = this.getSettings()
   512→
   513→    const ch = get<ChapterRow>('SELECT * FROM chapters WHERE id = ?', task.chapterId)
   514→    if (!ch) {
   515→      this.bumpProgress(task.jobId)
   516→      return
   517→    }
   518→    // skip if already scraped by someone else
   519→    if (ch.content && ch.status !== 'error' && ch.status !== 'pending') {
   520→      this.bumpProgress(task.jobId)
   521→      return
   522→    }
   523→    const novel = getNovel(task.novelId)
   524→    const adapter: SiteAdapter | null = adapterById(novel?.source_site ?? '') ?? detectAdapter(ch.source_url)
   525→    if (!adapter) {
   526→      run('UPDATE chapters SET status = ?, error = ? WHERE id = ?', 'error', 'Không nhận diện được site', ch.id)
   527→      this.bumpProgress(task.jobId)
   528→      return
   529→    }
   530→
   531→    try {
   532→      const { title, content } = await adapter.fetchChapter(ch.source_url, { bookUrl: novel?.source_url })
   533→      let text = cleanNovelText(content, adapter.id)
   534→      // drop a leading line that duplicates the chapter title
   535→      const paras = text.split('\n\n')
   536→      if (paras.length > 1) {
   537→        const first = normCompare(paras[0])
   538→        const t = normCompare(title)
   539→        if (t && (first === t || (t.length >= 6 && first.startsWith(t)))) {
   540→          text = paras.slice(1).join('\n\n')
   541→        }
   542→      }
   543→      if (text.replace(/\s/g, '').length < 20) throw new Error('Nội dung chương trống sau khi làm sạch')
   544→      run(
   545→        `UPDATE chapters SET title = ?, content = ?, status = 'scraped', error = NULL, scraped_at = ? WHERE id = ?`,
   546→        ch.title || title || `Chương ${ch.idx}`,
   547→        text,
   548→        now(),
   549→        ch.id
   550→      )
   551→      this.consecutiveErrors.delete(task.jobId)
   552→      const job = getJob(task.jobId)
   553→      if (ch.idx % 10 === 0 && job) {
   554→        logInfo(`Đã cào ${job.progress + 1}/${job.total} chương «${novel?.title ?? ''}» (chương ${ch.idx})`)
   555→      }
   556→      this.emitChapterRow(ch.id)
   557→    } catch (e) {
   558→      const msg = (e as Error).message ?? String(e)
   559→      run('UPDATE chapters SET status = ?, error = ? WHERE id = ?', 'error', msg, ch.id)
   560→      const cnt = (this.consecutiveErrors.get(task.jobId) ?? 0) + 1
   561→      this.consecutiveErrors.set(task.jobId, cnt)
   562→      logWarn(`Lỗi cào chương ${ch.idx} «${ch.title}»: ${msg}`)
   563→      this.emitChapterRow(ch.id)
   564→      if (cnt > CONSECUTIVE_ERROR_LIMIT) {
   565→        logError(`Quá ${CONSECUTIVE_ERROR_LIMIT} lỗi liên tiếp — tạm dừng job (site có thể đang chặn)`)
   566→        try {
   567→          this.pause(task.jobId, `Tạm dừng tự động: hơn ${CONSECUTIVE_ERROR_LIMIT} lỗi liên tiếp (site có thể đang chặn)`)
   568→        } catch {
   569→          /* ignore */
   570→        }
   571→        return
   572→      }
   573→    }
   574→
   575→    this.recomputeNovelCounts(task.novelId)
   576→    this.bumpProgress(task.jobId)
   577→
   578→    const delay = Math.max(0, s.scrapeDelayMs || 0)
   579→    if (delay) await sleep(delay)
   580→  }

   582→  private async processTranslateTask(task: TranslateTask): Promise<void> {
   583→    if (!this.ensureJobRunning(task.jobId)) return
   584→    const s = this.getSettings()
   585→
   586→    const ch = get<ChapterRow>('SELECT * FROM chapters WHERE id = ?', task.chapterId)
   587→    if (!ch || ch.content === null) {
   588→      this.bumpProgress(task.jobId)
   589→      return
   590→    }
   591→    const novel = getNovel(task.novelId)
   592→    if (!novel) {
   593→      this.bumpProgress(task.jobId)
   594→      return
   595→    }
   596→
   597→    run("UPDATE chapters SET status = 'translating' WHERE id = ?", ch.id)
   598→    this.emitChapterRow(ch.id)
   599→
   600→    const srcLang = s.sourceLang === 'auto' ? novel.lang : s.sourceLang
   601→    try {
   602→      const translated = await translateText(ch.content, {
   603→        novelTitle: novel.title,
   604→        novelAuthor: novel.author,
   605→        sourceLang: srcLang,
   606→        targetLang: s.targetLang,
   607→        model: s.geminiModel,
   608→        chunkSize: s.chunkSize,
   609→      })
   610→      if (!translated.trim()) throw new Error('Bản dịch rỗng')
   611→      run(
   612→        `UPDATE chapters SET translated = ?, status = 'translated', translated_at = ? WHERE id = ?`,
   613→        translated,
   614→        now(),
   615→        ch.id
   616→      )
   617→      this.consecutiveErrors.delete(task.jobId)
   618→      const job = getJob(task.jobId)
   619→      if (ch.idx % 10 === 0 && job) {
   620→        logInfo(`Đã dịch ${job.progress + 1}/${job.total} chương «${novel.title}» (chương ${ch.idx})`)
   621→      }
   622→    } catch (e) {
   623→      const err = e as Error
   624→      if (e instanceof GeminiNoKeysError) {
   625→        // no usable keys — abort the whole job, revert chapter
   626→        run("UPDATE chapters SET status = 'scraped' WHERE id = ?", ch.id)
   627→        run(
   628→          'UPDATE jobs SET status = ?, error = ?, finished_at = ?, updated_at = ? WHERE id = ?',
   629→          'error',
   630→          err.message,
   631→          now(),
   632→          now(),
   633→          task.jobId
   634→        )
   635→        this.dropQueuedTasks(task.jobId)
   636→        logError(`Job dịch dừng: ${err.message}`)
   637→        emitChapterRow(ch.id)
   638→        emitJob(task.jobId)
   639→        return
   640→      }
   641→      run('UPDATE chapters SET status = ?, error = ? WHERE id = ?', 'error', err.message ?? String(err), ch.id)
   642→      const cnt = (this.consecutiveErrors.get(task.jobId) ?? 0) + 1
   643→      this.consecutiveErrors.set(task.jobId, cnt)
   644→      logWarn(`Lỗi dịch chương ${ch.idx} «${ch.title}»: ${err.message}`)
   645→      if (cnt > CONSECUTIVE_ERROR_LIMIT) {
   646→        logError(`Quá ${CONSECUTIVE_ERROR_LIMIT} lỗi dịch liên tiếp — tạm dừng job`)
   647→        try {
   648→          this.pause(task.jobId, `Tạm dừng tự động: hơn ${CONSECUTIVE_ERROR_LIMIT} lỗi dịch liên tiếp`)
   649→        } catch {
   650→          /* ignore */
   651→        }
   652→        return
   653→      }
   654→    }
   655→
   656→    this.recomputeNovelCounts(task.novelId)
   657→    this.emitChapterRow(ch.id)
   658→    this.bumpProgress(task.jobId)
   659→
   660→    const delay = Math.max(0, s.translateDelayMs || 0)
   661→    if (delay) await sleep(delay)
   662→  }

   668→  private bumpProgress(jobId: string): void {
   669→    run(
   670→      "UPDATE jobs SET progress = MIN(progress + 1, total), updated_at = ? WHERE id = ? AND status IN ('running','queued')",
   671→      now(),
   672→      jobId
   673→    )
   674→    emitJob(jobId)
   675→  }

   677→  private maybeFinishJob(jobId: string): void {
   678→    if ((this.inFlight.get(jobId) ?? 0) > 0) return
   679→    if (this.scrapeQueue.some((t) => t.jobId === jobId)) return
   680→    if (this.translateQueue.some((t) => t.jobId === jobId)) return
   681→    const job = getJob(jobId)
   682→    if (!job || (job.status !== 'running' && job.status !== 'queued')) return
   683→
   684→    run('UPDATE jobs SET status = ?, error = NULL, finished_at = ?, updated_at = ? WHERE id = ?', 'done', now(), now(), jobId)
   685→    const novel = job.novel_id ? getNovel(job.novel_id) : null
   686→    logSuccess(
   687→      `Hoàn thành job ${job.type} «${novel?.title ?? '—'}»: ${job.progress}/${job.total} chương`
   688→    )
   689→    emitJob(jobId)
   690→
   691→    // post-actions for scrape/retry jobs
   692→    if ((job.type === 'scrape' || job.type === 'retry') && job.novel_id) {
   693→      let params: Record<string, unknown> = {}
   694→      try {
   695→        params = JSON.parse(job.params || '{}')
   696→      } catch {
   697→        params = {}
   698→      }
   699→      const s = this.getSettings()
   700→      if (params.autoTranslate === true) {
   701→        const n = getNovel(job.novel_id)
   702→        const srcLang = n ? (s.sourceLang === 'auto' ? n.lang : s.sourceLang) : 'zh'
   703→        if (srcLang === s.targetLang) {
   704→          logInfo('Bỏ qua tự động dịch: ngôn ngữ nguồn trùng ngôn ngữ đích')
   705→        } else if (!hasActiveGeminiKeys()) {
   706→          logWarn('Bỏ qua tự động dịch: chưa có Gemini API key hoạt động')
   707→        } else {
   708→          void this.startTranslateJob({ novelId: job.novel_id }).catch((e) =>
   709→            logError(`Không tạo được job dịch tự động: ${(e as Error).message}`)
   710→          )
   711→        }
   712→      }
   713→      if (s.autoSyncSupabase && s.supabaseEnabled && s.supabaseUrl && s.supabaseKey) {
   714→        void supabasePush(job.novel_id)
   715→          .then((r) => logSuccess(`Tự động sync Supabase xong: ${r.novels} truyện / ${r.chapters} chương`))
   716→          .catch((e) => logWarn(`Tự động sync Supabase lỗi (không ảnh hưởng job): ${(e as Error).message}`))
   717→      }
   718→    }
   719→  }

   721→  private recomputeNovelCounts(novelId: string): void {
   722→    run(
   723→      `UPDATE novels SET
   724→         scraped_count = (SELECT COUNT(*) FROM chapters WHERE novel_id = ? AND status IN ('scraped','translating','translated')),
   725→         translated_count = (SELECT COUNT(*) FROM chapters WHERE novel_id = ? AND status = 'translated'),
   726→         updated_at = ?
   727→       WHERE id = ?`,
   728→      novelId,
   729→      novelId,
   730→      now(),
   731→      novelId
   732→    )
   733→    const n = getNovel(novelId)
   734→    if (n) emitNovel(novelRowToSummary(n))
   735→  }

   737→  private emitChapterRow(chapterId: string): void {
   738→    const ch = get<ChapterRow>('SELECT * FROM chapters WHERE id = ?', chapterId)
   739→    if (!ch) return
   740→    emitChapter({
   741→      novelId: ch.novel_id,
   742→      idx: ch.idx,
   743→      status: ch.status,
   744→      hasContent: ch.content !== null && ch.content.length > 0,
   745→      hasTranslated: ch.translated !== null && ch.translated.length > 0,
   746→    })
   747→  }

   749→  buildStats(): EngineStats {
   750→    const row = get<Record<string, number>>(
   751→      `SELECT
   752→        (SELECT COUNT(*) FROM novels) AS novels,
   753→        (SELECT COUNT(*) FROM chapters) AS chapters,
   754→        (SELECT COUNT(*) FROM chapters WHERE status IN ('scraped','translating','translated')) AS scraped,
   755→        (SELECT COUNT(*) FROM chapters WHERE status = 'translated') AS translated,
   756→        (SELECT COUNT(*) FROM jobs WHERE status = 'running') AS runningJobs,
   757→        (SELECT COUNT(*) FROM jobs WHERE status = 'queued') AS queuedJobs,
   758→        (SELECT COUNT(*) FROM api_keys) AS keysTotal,
   759→        (SELECT COUNT(*) FROM api_keys WHERE active = 1 AND status != 'error') AS keysActive,
   760→        (SELECT COUNT(*) FROM api_keys WHERE status = 'rate_limited') AS keysRateLimited,
   761→        (SELECT COUNT(*) FROM api_keys WHERE status = 'error') AS keysError`
   762→    )
   763→    return {
   764→      novels: row?.novels ?? 0,
   765→      chapters: row?.chapters ?? 0,
   766→      scraped: row?.scraped ?? 0,
   767→      translated: row?.translated ?? 0,
   768→      runningJobs: row?.runningJobs ?? 0,
   769→      queuedJobs: row?.queuedJobs ?? 0,
   770→      keysTotal: row?.keysTotal ?? 0,
   771→      keysActive: row?.keysActive ?? 0,
   772→      keysRateLimited: row?.keysRateLimited ?? 0,
   773→      keysError: row?.keysError ?? 0,
   774→    }
   775→  }
